# =============================================================================
# Magic Frames — single-origin cPanel .htaccess
# Place this file as:  /home/uddjzwrz/album-magicframes.nokkoo.in/.htaccess
#
# Layout assumed:
#   DOC ROOT  = /home/uddjzwrz/album-magicframes.nokkoo.in
#     ├── index.html, assets/, favicon.svg, manifest.webmanifest   (React build)
#     ├── index.php, favicon.ico, robots.txt                       (Laravel public/)
#     └── .htaccess                                                (this file)
#   Laravel app (code + vendor + .env) lives OUTSIDE the doc root at:
#     /home/uddjzwrz/laravel/album-magicframes/backend
#
# Behaviour:
#   /.well-known/acme-challenge/* -> served over HTTP for AutoSSL validation
#   /api/*            -> handled by Laravel (index.php)
#   /up               -> Laravel health check
#   everything else   -> React SPA (index.html) for client-side routing
# =============================================================================

# ---- Disable LiteSpeed server-side cache for this site ----
# The host runs LiteSpeed, which was serving a STALE cached copy of index.html
# (ignoring the no-cache response header and ?query cache-busting). Explicitly
# turn LiteSpeed cache OFF so new deploys are always served immediately.
<IfModule LiteSpeed>
    CacheLookup off
    RewriteEngine On
    RewriteRule .* - [E=Cache-Control:no-cache]
    RewriteRule .* - [E=cache-control:no-cache]
</IfModule>

# ---- Serve the React SPA shell (index.html) for the directory root ----
# Apache/LiteSpeed default DirectoryIndex prefers index.php (the Laravel front
# controller), which has no route for "/" and returns an empty response. Force
# index.html so a bare "/" request loads the SPA. /api/* still reaches index.php
# via the RewriteRule below.
<IfModule mod_dir.c>
    DirectoryIndex index.html index.php
</IfModule>

<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # ---- Let's Encrypt / AutoSSL HTTP-01 validation ----
    RewriteRule ^\.well-known/acme-challenge/ - [END]

    # ---- API + health check go to the PHP front controller ----
    # These are checked first; they exit the rewrite pass so Laravel handles
    # them. Everything NOT matching these falls through to the SPA rules below.
    RewriteRule ^api/ index.php [L]
    RewriteRule ^up$ index.php [L]

    # ---- SPA fallback ----
    # Serve any real existing file (JS/CSS/images/index.php/etc.) as-is; for
    # every other path, return the React shell so client-side routing works.
    # NOTE for LiteSpeed: we match on REQUEST_FILENAME not being a real file and
    # rewrite to the absolute /index.html. The HTTPS redirect was removed from
    # here (AutoSSL/host already forces HTTPS) because on LiteSpeed its
    # interaction with the fallback was terminating the rewrite pass for deep
    # paths, causing /admin/* to 404.
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteRule ^ /index.html [L]
</IfModule>

# ---- Standard Laravel public/.htaccess directives (for index.php requests) ----
<IfModule mod_negotiation.c>
    Options -MultiViews -Indexes
</IfModule>

<IfModule mod_headers.c>
    # Long-cache fingerprinted build assets
    <FilesMatch "\.(js|css|woff2|svg|png|jpg|jpeg|webp|ico)$">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>
    # Never cache the SPA shell so new deploys are picked up
    <FilesMatch "^(index\.html|manifest\.webmanifest)$">
        Header set Cache-Control "no-cache, no-store, must-revalidate"
    </FilesMatch>
</IfModule>
