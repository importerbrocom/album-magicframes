# =============================================================================
# Magic Frames — single-origin cPanel .htaccess (LiteSpeed)
# Place this file as:  /home/uddjzwrz/album-magicframes.nokkoo.in/.htaccess
#
# Layout:
#   DOC ROOT  = /home/uddjzwrz/album-magicframes.nokkoo.in
#     ├── index.html, assets/, favicon.svg, manifest.webmanifest   (React build)
#     ├── index.php, favicon.ico, robots.txt                       (Laravel public/)
#     └── .htaccess                                                (this file)
#   Laravel app lives OUTSIDE the doc root at:
#     /home/uddjzwrz/laravel/album-magicframes/backend
#
# IMPORTANT — this vhost does NOT honor "RewriteRule ^ index.html" SPA
# fallbacks (LiteSpeed per-vhost rewrite restriction: even an explicit test
# rule did not fire, so /admin/* 404'd). But index.php IS always reached.
# So we route EVERYTHING non-file through the Laravel front controller, and
# Laravel serves the React shell (routes/web.php catch-all) for non-API paths.
# =============================================================================

# ---- Disable LiteSpeed server-side cache for this site ----
<IfModule LiteSpeed>
    CacheLookup off
    RewriteEngine On
    RewriteRule .* - [E=Cache-Control:no-cache]
</IfModule>

# ---- Directory index: hit index.php (Laravel) for "/" so the catch-all in
#      routes/web.php serves the SPA shell. Works even if rewrites are off. ----
<IfModule mod_dir.c>
    DirectoryIndex index.php index.html
</IfModule>

<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # Let's Encrypt / AutoSSL HTTP-01 validation.
    RewriteRule ^\.well-known/acme-challenge/ - [END]

    # Everything that is NOT a real file/dir -> Laravel front controller.
    # Laravel handles /api/* + /up, and serves the React SPA shell for every
    # other path. Real files (assets/*.js, *.css, images, favicon) are served
    # directly by the server and never reach this rule.
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteRule ^ index.php [L]
</IfModule>

<IfModule mod_negotiation.c>
    Options -MultiViews -Indexes
</IfModule>

<IfModule mod_headers.c>
    # Long-cache fingerprinted build assets.
    <FilesMatch "\.(js|css|woff2|svg|png|jpg|jpeg|webp|ico)$">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>
    # Never cache the SPA shell / manifest so new deploys are picked up.
    <FilesMatch "^(index\.html|manifest\.webmanifest)$">
        Header set Cache-Control "no-cache, no-store, must-revalidate"
    </FilesMatch>
</IfModule>
