STEP 1 — Add the controller

mkdir -p app/Http/Controllers/Api
# then create app/Http/Controllers/Api/LeadApiController.php with the

--------------------------------------------------------------------------
STEP 2 — Create routes/api.php  (it does NOT exist yet on this app)
--------------------------------------------------------------------------
# Create the file ~/magic.nokkoo.in/routes/api.php with EXACTLY:

<?php

use App\Http\Controllers\Api\LeadApiController;
use Illuminate\Support\Facades\Route;

Route::post('/leads', [LeadApiController::class, 'store']);

# Routes in api.php are auto-prefixed with /api, so this is POST /api/leads.

--------------------------------------------------------------------------
STEP 2b — Register the api file in bootstrap/app.php  (REQUIRED here)
--------------------------------------------------------------------------
# This app's withRouting() does NOT list an api: file. Edit bootstrap/app.php
# and add the api: line inside ->withRouting( ... ) so it reads:
#
#     ->withRouting(
#         web: __DIR__.'/../routes/web.php',
#         api: __DIR__.'/../routes/api.php',        <-- ADD THIS LINE
#         commands: __DIR__.'/../routes/console.php',
#         channels: __DIR__.'/../routes/channels.php',
#         health: '/up',
#     )
#
# Then clear route cache (STEP 5 does this).
#
# NOTE: api routes get the "api" middleware group automatically, which does
# NOT include the web group's CSRF / Inertia / Spatie middleware — so the
# album's token-only POST is accepted. No extra middleware config needed.

--------------------------------------------------------------------------
STEP 3 — Add the shared secret + system user to .env
--------------------------------------------------------------------------
# Generate a strong token:
openssl rand -hex 32
# Add these two lines to ~/magic.nokkoo.in/.env  (use the id from STEP 0d):
#   LEADS_API_TOKEN=<the-token-from-above>
#   LEADS_API_SYSTEM_USER_ID=<a-real-user-id>

--------------------------------------------------------------------------
STEP 4 — REQUIRED: expose them via config/services.php
--------------------------------------------------------------------------
# Required, not optional: magic caches its config, and env() returns null
# when cached (-> 401 on every call). The controller reads config() only.
# After a new class file also run: composer dump-autoload -o
# Add this block to the array in config/services.php:
#   'leads_api' => [
#       'token' => env('LEADS_API_TOKEN'),
#       'system_user_id' => env('LEADS_API_SYSTEM_USER_ID', 1),
#   ],

--------------------------------------------------------------------------
STEP 5 — Clear caches
--------------------------------------------------------------------------
/opt/alt/php83/usr/bin/php artisan config:clear
/opt/alt/php83/usr/bin/php artisan route:clear
# Confirm the route now exists:
/opt/alt/php83/usr/bin/php artisan route:list | grep -i "api/leads"

  -H "Authorization: Bearer TOKEN" \
  -H "Accept: application/json" \
